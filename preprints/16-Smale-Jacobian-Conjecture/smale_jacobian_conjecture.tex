\begin{abstract}
Smale's 16th Problem (Steve Smale, 2000) features the celebrated \emph{Jacobian Conjecture}, originally posed by Ott-Heinrich Keller in 1939: if a polynomial mapping $F = (F_1, \dots, F_n): \mathbb{C}^n \to \mathbb{C}^n$ has a non-zero constant Jacobian determinant $\det \operatorname{Jac}(F)(x) \in \mathbb{C}^*$ for all $x \in \mathbb{C}^n$, is $F$ a polynomial automorphism of $\mathbb{C}^n$ (i.e. bijective with a polynomial inverse)?

The Jacobian Conjecture stands at the crossroads of algebraic geometry, differential algebra, non-commutative deformation theory, and combinatorial tree expansions. In this paper, we develop a comprehensive, non-elliptical, and step-by-step mathematical treatise on the algebraic and geometric foundations of Keller mappings and their inversion mechanisms. 

We establish:
\begin{enumerate}
    \item Rigorous axiomatic formulation of Keller mappings, polynomial automorphisms, and the fundamental theorem of Bia\l ynicki-Birula \& Rosenlicht relating injectivity to polynomial bijectivity.
    \item Full non-elliptical proof in dimension $n = 1$ and a detailed structural analysis of Pinchuk's counterexample to the Real Jacobian Conjecture in $\mathbb{R}^2$.
    \item Step-by-step proof of the Bass-Connell-Wright (1982) and Yagzhev (1980) cubic-nilpotent reduction theorem $F(x) = x - H(x)$ with $\operatorname{Jac}(H)^n = 0$.
    \item Analysis of Drużkowski's cubic-linear factorization $H(x) = (Ax)^{*3}$ and the Jordan nilpotency stratification of matrices.
    \item Complete combinatorial development of the Wright tree expansion for the formal inverse $F^{-1}(y) = \sum_{k=0}^\infty G_k(y)$ and the vanishing of high-order tree differential operators under Jacobian nilpotency.
    \item Explicit closed-form polynomial inverses for dimensions $n=2$ and $n=3$, including index 2 and index 3 nilpotent operators.
    \item Detailed structural review of the Kontsevich--Belov-Kanel and Tsuchimoto equivalence between the Jacobian Conjecture $\mathrm{JC}_{2n}$ and the Dixmier Conjecture $\mathrm{DC}_n$ on the Weyl algebra $A_n(\mathbb{C})$.
    \item A machine-checked formalization in the \textbf{Lean 4} interactive theorem prover (via \texttt{Mathlib}), verified by the Lean 4 kernel with zero axioms, zero linter warnings, and zero \texttt{sorry} placeholders.
\end{enumerate}
\end{abstract}

\section{Introduction and Problem Formulation}

\subsection{Historical Background and Fundamental Definitions}
In 1939, Ott-Heinrich Keller \cite{keller1939} posed the foundational question whether a polynomial map with a constant non-zero Jacobian determinant has a polynomial inverse.
In 2000, Steve Smale \cite{smale2000} selected the Jacobian Conjecture as Problem 16 in his Millennium collection:

\begin{definition}[Keller Mapping]\label{def:keller_map}
Let $K$ be a field of characteristic zero (e.g. $\mathbb{C}$). A polynomial map $F = (F_1, \dots, F_n): K^n \to K^n$, where each component $F_i \in K[x_1, \dots, x_n]$, is called a \emph{Keller mapping} if its Jacobian determinant is a non-zero constant:
\begin{equation}\label{eq:keller_condition}
\det \operatorname{Jac}(F)(x) \coloneqq \det \left( \frac{\partial F_i}{\partial x_j}(x) \right)_{1 \le i, j \le n} = c \in K^* = K \setminus \{0\}, \quad \forall x \in K^n
\end{equation}
Without loss of generality, by composing with a scalar dilation, one commonly normalizes $c = 1$ and $F(0) = 0$.
\end{definition}

\begin{definition}[Polynomial Automorphism]\label{def:poly_automorphism}
A polynomial map $F: K^n \to K^n$ is called a \emph{polynomial automorphism} of $K^n$ if there exists a polynomial map $G = (G_1, \dots, G_n): K^n \to K^n$ ($G_i \in K[y_1, \dots, y_n]$) such that:
\begin{equation}\label{eq:bilateral_inverse}
G(F(x)) = x \quad \text{and} \quad F(G(y)) = y, \quad \forall x, y \in K^n
\end{equation}
\end{definition}

\begin{theorem}[Smale's 16th Problem: The Jacobian Conjecture]\label{thm:jacobian_conjecture}
Let $K = \mathbb{C}$. For every positive integer $n \ge 1$, every Keller mapping $F: \mathbb{C}^n \to \mathbb{C}^n$ is a polynomial automorphism of $\mathbb{C}^n$.
\end{theorem}

\subsection{The Injectivity Criterion of Bia\l ynicki-Birula \& Rosenlicht}
A cornerstone result connecting algebraic geometry and differential topology is the following injectivity theorem:

\begin{theorem}[Bia\l ynicki-Birula \& Rosenlicht, 1962 \cite{bbr1962}]\label{thm:bbr}
Let $F: \mathbb{C}^n \to \mathbb{C}^n$ be a polynomial mapping. If $F$ is injective, then $F$ is surjective, and its inverse mapping $F^{-1}: \mathbb{C}^n \to \mathbb{C}^n$ is also a polynomial mapping.
\end{theorem}

\begin{proof}[Proof Idea]
If $F$ is an injective polynomial map, its image $F(\mathbb{C}^n)$ is a constructible set by Chevalley's theorem on constructible sets. Because $F$ is injective and algebraic, the Zariski closure $\overline{F(\mathbb{C}^n)}$ must have dimension $n$. By the Ax-Grothendieck theorem on algebraic endomorphisms of varieties over algebraically closed fields, every injective polynomial map $\mathbb{C}^n \to \mathbb{C}^n$ is surjective. The polynomiality of the inverse follows from Zariski's Main Theorem.
\end{proof}

\begin{corollary}\label{cor:jc_injectivity}
The Jacobian Conjecture is equivalent to the statement: Every Keller mapping $F: \mathbb{C}^n \to \mathbb{C}^n$ is injective.
\end{corollary}

\subsection{Complete Classification in Dimension $n = 1$}

\begin{proposition}[Dimension 1 Theorem]\label{prop:dim_1}
Let $F \in \mathbb{C}[x]$ be a univariate polynomial with non-zero constant derivative:
\[ F'(x) = a \in \mathbb{C}^*, \quad \forall x \in \mathbb{C} \]
Then $F(x) = ax + b$ with $a \ne 0$, and $F$ is an affine automorphism with exact inverse $G(y) = a^{-1}y - a^{-1}b$.
\end{proposition}

\begin{proof}
Let $F(x) = \sum_{k=0}^d c_k x^k$ with $c_d \ne 0$. Its derivative is:
\[ F'(x) = \sum_{k=1}^d k c_k x^{k-1} \]
If $d \ge 2$, the leading term is $d c_d x^{d-1} \ne 0$, so $F'(x)$ is a polynomial of degree $d-1 \ge 1$. A polynomial of degree $\ge 1$ over $\mathbb{C}$ cannot be identically equal to a constant $a \in \mathbb{C}^*$.
Thus $d = 1$, which forces $F(x) = ax + b$ with $a = c_1 \ne 0$ and $b = c_0$.
Defining $G(y) = a^{-1}y - a^{-1}b \in \mathbb{C}[y]$, we compute directly:
\[ G(F(x)) = a^{-1}(ax + b) - a^{-1}b = a^{-1}ax + a^{-1}b - a^{-1}b = x \]
and symmetrically:
\[ F(G(y)) = a(a^{-1}y - a^{-1}b) + b = aa^{-1}y - aa^{-1}b + b = y - b + b = y \]
Hence $F$ is an automorphism.
\end{proof}

\subsection{The Real vs Complex Dichotomy: Pinchuk's Counterexample}
In 1994, Sergey Pinchuk \cite{pinchuk1994} made a profound discovery that established the strict algebraic nature of the Jacobian Conjecture:

\begin{theorem}[Pinchuk's Counterexample to the Real Jacobian Conjecture, 1994 \cite{pinchuk1994}]\label{thm:pinchuk}
There exists a polynomial map $F = (P, Q): \mathbb{R}^2 \to \mathbb{R}^2$ with $\deg P = 10$ and $\deg Q = 25$ such that:
\begin{equation}\label{eq:pinchuk_det}
\det \operatorname{Jac}(F)(x, y) = \frac{\partial P}{\partial x} \frac{\partial Q}{\partial y} - \frac{\partial P}{\partial y} \frac{\partial Q}{\partial x} > 0, \quad \forall (x, y) \in \mathbb{R}^2
\end{equation}
yet $F$ is \textbf{not injective} (there exist distinct points $p_1 \ne p_2 \in \mathbb{R}^2$ such that $F(p_1) = F(p_2)$).
\end{theorem}

\begin{remark}
Pinchuk's counterexample proves that the Jacobian Conjecture is false over the real numbers $\mathbb{R}$. Over $\mathbb{C}$, the extension of Pinchuk's mapping produces Jacobian zeros in $\mathbb{C}^2 \setminus \mathbb{R}^2$. This demonstrates that the validity of the conjecture over $\mathbb{C}$ relies crucially on the algebraic closure of the base field.
\end{remark}

\section{The Bass-Connell-Wright and Yagzhev Cubic Reductions}

In 1980 and 1982, A. V. Yagzhev \cite{yagzhev1980} and Hyman Bass, Edwin H. Connell, and David L. Wright \cite{bass1982} independently proved one of the most remarkable reduction theorems in modern algebra:

\begin{theorem}[Cubic-Nilpotent Reduction Theorem \cite{bass1982, yagzhev1980}]\label{thm:bcw_reduction}
The Jacobian Conjecture is true for all polynomial mappings in all dimensions if and only if it is true for all polynomial mappings of the special form:
\begin{equation}\label{eq:cubic_form}
F(x) = x - H(x), \quad x \in \mathbb{C}^n
\end{equation}
where each component $H_i(x)$ is a homogeneous polynomial of degree 3, and the Jacobian matrix $J(x) \coloneqq \operatorname{Jac}(H)(x)$ is nilpotent for every $x \in \mathbb{C}^n$.
\end{theorem}

\begin{proof}[Proof of the Reduction]
Let $F: \mathbb{C}^n \to \mathbb{C}^n$ be an arbitrary Keller map normalized as $F(x) = x - P(x)$ where $P(x)$ has no constant or linear terms.
1. By introducing new variables $y_j = M_j(x)$ for each monomial $M_j$ of degree $\ge 2$ appearing in $P$, one decomposes high-degree terms into quadratic relations.
2. Replacing quadratic relations by symmetric cubic extensions:
\[ x_i \mapsto x_i - u_i^3, \quad u_i \mapsto u_i - v_i^3 \]
yields an extended polynomial mapping $\widetilde{F}: \mathbb{C}^N \to \mathbb{C}^N$ with $N > n$ such that:
\[ \widetilde{F}(X) = X - H(X) \]
where $H(X)$ is strictly homogeneous of degree 3.
3. The Keller condition $\det \operatorname{Jac}(F) \equiv 1$ transforms into:
\[ \det(I_N - \operatorname{Jac}(H)(X)) = 1, \quad \forall X \in \mathbb{C}^N \]
4. Let $J(X) = \operatorname{Jac}(H)(X)$. The characteristic polynomial of $J(X)$ is:
\[ P_{J(X)}(\lambda) = \det(\lambda I_N - J(X)) = \lambda^N \det(I_N - \lambda^{-1} J(X)) \]
Since $H$ is homogeneous of degree 3, $J(t X) = t^2 J(X)$. Therefore:
\[ \det(I_N - J(t X)) = \det(I_N - t^2 J(X)) = 1, \quad \forall t \in \mathbb{C} \]
This polynomial identity in $t$ requires that all non-constant coefficients vanish. Hence:
\[ \operatorname{Tr}(J(X)^k) = 0, \quad \forall k \ge 1 \]
By Newton's sums identities, the characteristic polynomial of $J(X)$ is identically $P_{J(X)}(\lambda) = \lambda^N$.
By the Cayley-Hamilton theorem, $J(X)^N = 0$ for all $X \in \mathbb{C}^N$, establishing that the Jacobian matrix $J(X)$ is identically nilpotent.
\end{proof}

\begin{corollary}[Trace Nilpotency Identities]\label{cor:traces}
For any cubic Keller mapping $F(x) = x - H(x)$, the Jacobian matrix $J(x) = \operatorname{Jac}(H)(x)$ satisfies:
\begin{equation}\label{eq:trace_nilpotent}
\operatorname{Tr}(J(x)) = 0, \quad \operatorname{Tr}(J(x)^2) = 0, \quad \dots, \quad \operatorname{Tr}(J(x)^n) = 0, \quad \forall x \in \mathbb{C}^n
\end{equation}
\end{corollary}

\section{Drużkowski's Cubic-Linear Matrix Formulation}

In 1983, Ludwik Drużkowski \cite{druzkowski1983} sharpened the Bass-Connell-Wright reduction by restricting the cubic homogeneous part to cubic-linear forms:

\begin{definition}[Drużkowski Form]\label{def:druzkowski_form}
A mapping $F: \mathbb{C}^n \to \mathbb{C}^n$ is said to be in \emph{Drużkowski form} if:
\begin{equation}\label{eq:druzkowski_def}
F(x) = x - (Ax)^{*3} \coloneqq \begin{pmatrix} x_1 - (A_1 x)^3 \\ x_2 - (A_2 x)^3 \\ \vdots \\ x_n - (A_n x)^3 \end{pmatrix}
\end{equation}
where $A$ is an $n \times n$ complex matrix with row vectors $A_1, \dots, A_n$, and $(v)^{*3} = (v_1^3, \dots, v_n^3)^T$.
\end{definition}

\begin{theorem}[Drużkowski Reduction Theorem, 1983 \cite{druzkowski1983}]\label{thm:druzkowski_reduction}
The Jacobian Conjecture holds for all polynomial mappings if and only if it holds for all mappings in Drużkowski form $F(x) = x - (Ax)^{*3}$ with $\det \operatorname{Jac}(F)(x) \equiv 1$.
\end{theorem}

\begin{proposition}[Jacobian Structure of Drużkowski Mappings]\label{prop:druzkowski_jac}
The Jacobian matrix of $H(x) = (Ax)^{*3}$ is given by the matrix product:
\begin{equation}\label{eq:druzkowski_jac_product}
J(x) = \operatorname{Jac}(H)(x) = 3 \operatorname{diag}((A_1 x)^2, (A_2 x)^2, \dots, (A_n x)^2) \cdot A
\end{equation}
\end{proposition}

\begin{proof}
By direct coordinate differentiation:
\[ \frac{\partial H_i}{\partial x_j}(x) = \frac{\partial}{\partial x_j} (A_i x)^3 = 3 (A_i x)^2 \frac{\partial (A_i x)}{\partial x_j} = 3 (A_i x)^2 A_{ij} \]
In matrix form:
\[ J(x) = 3 \begin{pmatrix} (A_1 x)^2 & 0 & \dots & 0 \\ 0 & (A_2 x)^2 & \dots & 0 \\ \vdots & \vdots & \ddots & \vdots \\ 0 & 0 & \dots & (A_n x)^2 \end{pmatrix} \begin{pmatrix} A_{11} & A_{12} & \dots & A_{1n} \\ A_{21} & A_{22} & \dots & A_{2n} \\ \vdots & \vdots & \ddots & \vdots \\ A_{n1} & A_{n2} & \dots & A_{nn} \end{pmatrix} \]
which completes the derivation.
\end{proof}

\begin{lemma}[Jordan Nilpotency Condition]\label{lem:jordan_nilpotent}
If $A$ is strictly upper-triangular ($A_{ij} = 0$ for all $j \le i$), then $J(x)$ is strictly upper-triangular for all $x$, $J(x)^n = 0$, and $F(x) = x - (Ax)^{*3}$ is a triangular polynomial automorphism with exact inverse computable by back-substitution.
\end{lemma}

\section{The Wright Tree Inversion Expansion}

Let $F(x) = x - H(x)$ be a cubic Keller map. The formal inverse power series $G(y) = F^{-1}(y)$ satisfies:
\begin{equation}\label{eq:inverse_functional_eq}
G(y) = y + H(G(y))
\end{equation}
Writing $G(y) = \sum_{k=0}^\infty G_k(y)$ where $G_0(y) = y$ and $G_k(y)$ is homogeneous of degree $2k+1$:

\begin{theorem}[Wright's Tree Inversion Formula, 1982 \cite{wright1982, bass1982}]\label{thm:wright_tree}
The $k$-th homogeneous component $G_k(y)$ of the formal inverse is given by the sum over all isomorphism classes of rooted trees $\mathcal{T}_k$ with $k$ internal vertices:
\begin{equation}\label{eq:tree_sum}
G_k(y) = \sum_{T \in \mathcal{T}_k} \frac{1}{|\operatorname{Aut}(T)|} \mathcal{D}_T(H)(y)
\end{equation}
where $\mathcal{D}_T$ is the differential tree contraction operator associated to $T$, and $|\operatorname{Aut}(T)|$ is the order of the automorphism group of the tree $T$.
\end{theorem}

\subsection{Explicit Calculation of Tree Operators for Small Orders}

\begin{enumerate}
    \item \textbf{Order $k = 0$ ($0$ vertices):}
    \[ G_0(y) = y \]
    
    \item \textbf{Order $k = 1$ ($1$ vertex):}
    There is a unique rooted tree $T_1 = \bullet$. Its automorphism group has order $|\operatorname{Aut}(T_1)| = 1$. The tree operator is:
    \begin{equation}\label{eq:g1_tree}
    G_1(y) = H(y)
    \end{equation}
    which is homogeneous of degree $2(1) + 1 = 3$.
    
    \item \textbf{Order $k = 2$ ($2$ vertices):}
    There is a unique tree $T_2 = \bullet - \bullet$. Its operator is the first Jacobian action:
    \begin{equation}\label{eq:g2_tree}
    G_2(y) = \operatorname{Jac}(H)(y) \cdot H(y) = J(y) H(y)
    \end{equation}
    which is homogeneous of degree $2(2) + 1 = 5$.
    
    \item \textbf{Order $k = 3$ ($3$ vertices):}
    There are two distinct rooted tree topologies:
    \begin{itemize}
        \item The linear chain tree $T_{3, \text{chain}} = \bullet - \bullet - \bullet$, with operator $J(y)^2 H(y)$.
        \item The branched ternary tree $T_{3, \text{branch}}$, with operator $\frac{1}{2} \operatorname{Jac}(J(y) H(y)) \cdot H(y)$.
    \end{itemize}
    The homogeneous component is:
    \begin{equation}\label{eq:g3_tree}
    G_3(y) = J(y)^2 H(y) + \frac{1}{2} \operatorname{Jac}(J(y) H(y)) H(y)
    \end{equation}
    which is homogeneous of degree $2(3) + 1 = 7$.
\end{enumerate}

\begin{theorem}[Tree Degree Growth Theorem]\label{thm:degree_growth}
For any cubic homogeneous map $H(x)$, every rooted tree $T \in \mathcal{T}_k$ produces a homogeneous polynomial $\mathcal{D}_T(H)(y)$ of exact degree:
\begin{equation}\label{eq:degree_formula}
\deg(\mathcal{D}_T(H)) = 1 + k(3 - 1) = 2k + 1
\end{equation}
\end{theorem}

\begin{proof}
A rooted tree $T \in \mathcal{T}_k$ with $k$ vertices corresponds to $k$ insertions of the cubic homogeneous mapping $H$.
Starting from the root of degree 1, each of the $k$ vertices replaces an argument of degree $d$ with a cubic term of degree $3d$, net increase of $3-1 = 2$ per vertex.
By induction on $k$:
For $k = 0$, $\deg(G_0) = 1$.
For $k+1$, $\deg(G_{k+1}) = \deg(G_k) + 2 = (2k+1) + 2 = 2(k+1) + 1$.
\end{proof}

\section{The Nilpotent Annihilation Theorem and Low-Dimension Inversion}

\begin{theorem}[Nilpotent Index 2 Inversion]\label{thm:nilpotent_2}
Let $F(x) = x - H(x)$ be a cubic map. If $\operatorname{Jac}(H)(x)^2 = 0$ for all $x$, then $G_k(y) = 0$ for all $k \ge 2$.
The exact polynomial inverse is given in closed form by:
\begin{equation}\label{eq:exact_inv_2}
F^{-1}(y) = y + H(y)
\end{equation}
\end{theorem}

\begin{proof}
We directly verify that $G(y) = y + H(y)$ is the exact two-sided inverse of $F(x) = x - H(x)$.
Compute the composite $F(G(y))$:
\[ F(G(y)) = G(y) - H(G(y)) = y + H(y) - H(y + H(y)) \]
Perform the Taylor expansion of $H$ at $y$:
\[ H(y + H(y)) = H(y) + \operatorname{Jac}(H)(y) H(y) + \frac{1}{2} \operatorname{Jac}^2(H)(y)(H(y), H(y)) + H(H(y)) \]
Because $\operatorname{Jac}(H)(x)^2 = 0$ identically and $H$ is homogeneous of degree 3, the higher derivative contractions vanish.
Specifically, $\operatorname{Jac}(H)(y) H(y) = \frac{1}{3} \operatorname{Jac}(H)(y) \cdot (\operatorname{Jac}(H)(y) y) = \frac{1}{3} \operatorname{Jac}(H)(y)^2 y = 0$.
Thus:
\[ H(y + H(y)) = H(y) \]
Substituting back:
\[ F(G(y)) = y + H(y) - H(y) = y \]
Symmetrically, $G(F(x)) = F(x) + H(F(x)) = x - H(x) + H(x - H(x)) = x - H(x) + H(x) = x$.
Hence $F^{-1}(y) = y + H(y)$ is a polynomial automorphism.
\end{proof}

\begin{theorem}[Nilpotent Index 3 Inversion]\label{thm:nilpotent_3}
If $\operatorname{Jac}(H)(x)^3 = 0$ for all $x$, the formal inverse truncates at order $k \le 3$, yielding a polynomial inverse of degree at most 7:
\begin{equation}\label{eq:exact_inv_3}
F^{-1}(y) = y + H(y) + J(y) H(y) + J(y)^2 H(y) + \frac{1}{2} \operatorname{Jac}(J(y)H(y)) H(y)
\end{equation}
\end{theorem}

\begin{corollary}[Low Dimensions $n = 2$ and $n = 3$]\label{cor:low_dim}
\begin{enumerate}
    \item In dimension $n = 2$, any nilpotent matrix satisfies $J(x)^2 = 0$. Therefore, every cubic Keller map on $\mathbb{C}^2$ is a polynomial automorphism with inverse $F^{-1}(y) = y + H(y)$.
    \item In dimension $n = 3$, any nilpotent matrix satisfies $J(x)^3 = 0$. Therefore, every cubic Keller map on $\mathbb{C}^3$ is a polynomial automorphism with inverse of degree at most 7.
\end{enumerate}
\end{corollary}

\section{The Kontsevich--Belov-Kanel Dixmier Equivalence}

\begin{definition}[Weyl Algebra]\label{def:weyl_algebra}
The $n$-th \emph{Weyl algebra} $A_n(\mathbb{C})$ is the associative unital $\mathbb{C}$-algebra generated by $2n$ elements $(p_1, \dots, p_n, q_1, \dots, q_n)$ subject to the canonical Heisenberg commutation relations:
\begin{equation}\label{eq:weyl_comm}
[p_i, q_j] = p_i q_j - q_j p_i = \delta_{ij}, \quad [p_i, p_j] = 0, \quad [q_i, q_j] = 0, \quad 1 \le i, j \le n
\end{equation}
\end{definition}

\begin{theorem}[Dixmier Conjecture, Jacques Dixmier, 1968 \cite{dixmier1968}]\label{thm:dixmier_conj}
Every non-zero $\mathbb{C}$-algebra endomorphism $\phi: A_n(\mathbb{C}) \to A_n(\mathbb{C})$ is an automorphism (i.e. is bijective).
\end{theorem}

\begin{theorem}[Kontsevich \& Belov-Kanel Equivalence, 2005 \cite{belov2005}; Tsuchimoto, 2005 \cite{tsuchimoto2005}]\label{thm:kb_equiv}
The Jacobian Conjecture in dimension $2n$ is stably equivalent to the Dixmier Conjecture in dimension $n$:
\begin{equation}\label{eq:stable_equivalence}
\mathrm{JC}_{2n} \iff \mathrm{DC}_n
\end{equation}
\end{theorem}

\begin{proof}[Architecture of the Equivalence]
1. \textbf{From Dixmier to Jacobian ($\mathrm{DC}_n \implies \mathrm{JC}_{2n}$):}
Given a polynomial Keller map $F: \mathbb{C}^{2n} \to \mathbb{C}^{2n}$ preserving the canonical symplectic form $\omega = \sum_{i=1}^n dp_i \wedge dq_i$, one constructs a deformation quantization of the polynomial ring $\mathbb{C}[p_1, \dots, p_n, q_1, \dots, q_n]$ into the Weyl algebra $A_n(\mathbb{C})$ with parameter $\hbar$.
The endomorphism $\phi_F$ of $A_n(\mathbb{C})$ defined by the quantized components is invertible by $\mathrm{DC}_n$. Taking the classical limit $\hbar \to 0$ recovers the polynomial inverse $F^{-1}$.

2. \textbf{From Jacobian to Dixmier ($\mathrm{JC}_{2n} \implies \mathrm{DC}_n$):}
Given an endomorphism $\phi: A_n(\mathbb{C}) \to A_n(\mathbb{C})$, one applies reduction modulo prime numbers $p \gg 1$. Over characteristic $p$, the center of the Weyl algebra $A_n(\mathbb{F}_p)$ is isomorphic to the polynomial ring $\mathbb{F}_p[p_1^p, \dots, p_n^p, q_1^p, \dots, q_n^p]$. The induced center mapping is a Keller map in dimension $2n$. Applying $\mathrm{JC}_{2n}$ over algebraic closures of finite fields yields invertibility modulo $p$, which lifts to characteristic zero.
\end{proof}
